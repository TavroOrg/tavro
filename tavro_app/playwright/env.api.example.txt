# Backend API base URL under test — tavro_api has no JWT verification of its
# own (see tavro_api/api/dependencies.py::require_tenant), so once these
# tests are logged in as a real ZITADEL user, they hit tavro_api directly
# with that user's real access token + tenant id.
E2E_API_URL=http://localhost:8000

# Frontend URL — used only to drive the real ZITADEL hosted-UI login flow
# for the two provisioned UAT tenant users (see tests/setup/loginFlow.ts).
E2E_BASE_URL=http://localhost:9000

# ZITADEL Management API — used once at the start of the run to provision
# (or reuse, on subsequent runs) two real orgs + two real active human
# users acting as the two UAT tenants. See tests/setup/global-setup.ts.
# Not used at all if TENANT_A_USERNAME / TENANT_B_USERNAME are set below.
ZITADEL_API_URL=http://localhost:8080

# MCP server endpoint (mcp_server/server.py) — used only by the "MCP tools"
# section of api-tests.spec.ts.
MCP_URL=http://localhost:9001/zitadel/mcp

# ── Provisioning: pick ONE of the two modes below ──────────────────────────

# Mode A — auto-provision (local docker compose stack only). Pick ONE way
# to supply the ZITADEL admin credentials:
#
#   A1. Name of the running container that has the ZITADEL bootstrap admin
#       PAT mounted at /zitadel/bootstrap/admin.pat (the same PAT
#       iam/configure-zitadel-app.mjs uses). Only works if this machine has
#       local `docker exec` access to that container.
ZITADEL_BOOTSTRAP_CONTAINER=tavro-zitadel-configure-app-1
#
#   A2. A Personal Access Token for a ZITADEL IAM_OWNER machine user,
#       generated via the ZITADEL console (Users > Service Users > your
#       admin user > Personal Access Tokens > New). Takes priority over A1
#       if set. Use this for CI or any environment without docker exec
#       access. NOTE: this token can create/delete arbitrary orgs — do not
#       use this mode against production.
# ZITADEL_ADMIN_PAT=

# TENANT_A_USERNAME=
